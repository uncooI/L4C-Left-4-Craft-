cmake_minimum_required(VERSION 3.25)
project(Left4Craft LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(protocol_test tests/protocol_test.cpp)
target_include_directories(protocol_test PRIVATE ../protocol src)
enable_testing()
add_test(NAME protocol_coordinates COMMAND protocol_test)
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "L4D1 needs a Win32/x86 build, not x64.")
    endif()
    if(NOT MSVC)
        message(FATAL_ERROR "Source game interfaces require Microsoft's C++ ABI. Use Visual Studio Win32.")
    endif()
    set(HL2SDK "" CACHE PATH "Pinned AlliedModders hl2sdk l4d checkout")
    if(NOT EXISTS "${HL2SDK}/public/engine/iserverplugin.h")
        message(FATAL_ERROR "Set -DHL2SDK=... (scripts/build.ps1 fetches the pinned SDK)")
    endif()
    add_library(left4craft SHARED src/plugin.cpp src/integration.cpp src/renderer.cpp
        vendor/minhook/src/buffer.c vendor/minhook/src/hook.c vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde32.c)
    target_include_directories(left4craft PRIVATE ../protocol vendor/minhook/include "${HL2SDK}" "${HL2SDK}/public" "${HL2SDK}/public/tier0" "${HL2SDK}/public/tier1")
    target_compile_definitions(left4craft PRIVATE _WCHAR_T_DEFINED NDEBUG)
    target_link_libraries(left4craft PRIVATE shell32 d3d9 advapi32 "${HL2SDK}/lib/public/tier0.lib")
    set_property(TARGET left4craft PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
